import { Head } from '@inertiajs/react';

import ClientLayout from '@/layouts/client-layout';
import {
    LiveTrackingCard,
    type TrackingStage,
} from '@/components/client/live-tracking-card';
import {
    RequestRow,
    type ServiceRequestSummary,
} from '@/components/client/request-row';
import { ServiceGrid } from '@/components/client/service-grid';

type Props = {
    firstName: string;
    initials: string;
    activeTracking: TrackingStage | null;
    nextVisits: ServiceRequestSummary[];
};

/**
 * Client mobile home. On `<md` this stacks: greeting, the live-tracking hero
 * (when a technician is en route), quick-book service grid, and next visits.
 * The sidebar/topbar in ClientLayout take over at `md:` and up.
 */
export default function MobileHome({
    firstName,
    initials,
    activeTracking,
    nextVisits,
}: Props) {
    return (
        <ClientLayout>
            <Head title="Home" />

            <div className="mx-auto flex max-w-md flex-col gap-[18px] pb-6">
                <header className="flex items-end justify-between gap-3 px-5 pb-1 pt-2">
                    <div>
                        <p className="text-muted-foreground text-xs font-medium">
                            Good afternoon
                        </p>
                        <h1 className="mt-0.5 text-2xl font-bold leading-tight tracking-tight">
                            {firstName}
                        </h1>
                    </div>
                    <span className="bg-brand-soft text-brand-strong flex size-[38px] items-center justify-center rounded-full text-[13px] font-bold">
                        {initials}
                    </span>
                </header>

                {activeTracking ? (
                    <div className="px-5">
                        <LiveTrackingCard stage={activeTracking} />
                    </div>
                ) : null}

                <section>
                    <div className="flex items-center justify-between px-5 pb-2.5">
                        <h2 className="text-base font-bold">Book a service</h2>
                        <span className="text-brand-strong text-[13px] font-semibold">
                            All
                        </span>
                    </div>
                    <div className="px-5">
                        <ServiceGrid />
                    </div>
                </section>

                {nextVisits.length > 0 ? (
                    <section>
                        <div className="flex items-center justify-between px-5 pb-2.5">
                            <h2 className="text-base font-bold">Next visits</h2>
                            <span className="text-brand-strong text-[13px] font-semibold">
                                See all
                            </span>
                        </div>
                        <div className="flex flex-col gap-2.5 px-5">
                            {nextVisits.map((request) => (
                                <RequestRow key={request.id} request={request} />
                            ))}
                        </div>
                    </section>
                ) : null}
            </div>
        </ClientLayout>
    );
}
