import type { LucideIcon } from 'lucide-react';
import {
    Grid3x3,
    Sparkles,
    WashingMachine,
    Wind,
    Wrench,
    Zap,
} from 'lucide-react';

import { cn } from '@/lib/utils';

export type ServiceTile = {
    key: string;
    label: string;
    icon: LucideIcon;
};

/** Default catalog shortcuts — swap for your seeded `service_categories`. */
export const DEFAULT_SERVICES: ServiceTile[] = [
    { key: 'ac', label: 'AC service', icon: Wind },
    { key: 'cleaning', label: 'Cleaning', icon: Sparkles },
    { key: 'plumbing', label: 'Plumbing', icon: Wrench },
    { key: 'electrical', label: 'Electrical', icon: Zap },
    { key: 'appliance', label: 'Appliance', icon: WashingMachine },
    { key: 'more', label: 'More', icon: Grid3x3 },
];

type Props = {
    services?: ServiceTile[];
    onSelect?: (tile: ServiceTile) => void;
    className?: string;
};

/** 3-column quick-book grid of service categories. */
export function ServiceGrid({ services = DEFAULT_SERVICES, onSelect, className }: Props) {
    return (
        <div className={cn('grid grid-cols-3 gap-2.5', className)}>
            {services.map((tile) => {
                const Icon = tile.icon;
                return (
                    <button
                        key={tile.key}
                        type="button"
                        onClick={() => onSelect?.(tile)}
                        className="bg-card flex flex-col items-center gap-2 rounded-lg border px-2 py-3.5"
                    >
                        <span className="bg-brand-soft text-brand-strong flex size-[42px] items-center justify-center rounded-xl">
                            <Icon className="size-5" />
                        </span>
                        <span className="text-[12.5px] font-semibold">{tile.label}</span>
                    </button>
                );
            })}
        </div>
    );
}
