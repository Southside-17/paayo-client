import { BadgeCheck, MessageCircle, Phone } from 'lucide-react';

import { cn } from '@/lib/utils';

export type TrackingStage = {
    /** 0–1 fraction of the trip completed, drives the progress dashes. */
    progress: number;
    etaMinutes: number;
    serviceLabel: string;
    requestRef: string;
    destination: string;
    technician: { name: string; initials: string; team: string; idVerified?: boolean };
};

type Props = {
    stage: TrackingStage;
    className?: string;
    onOpen?: () => void;
    onCall?: () => void;
    onMessage?: () => void;
};

const DASHES = 5;

/**
 * "On the way" live-tracking hero used at the top of the client mobile home.
 * Brand-green gradient header with a spark (orange) live dot, ETA, progress
 * dashes, and a technician row with call / message actions.
 */
export function LiveTrackingCard({ stage, className, onOpen, onCall, onMessage }: Props) {
    const filled = Math.round(stage.progress * DASHES);

    return (
        <div
            className={cn(
                'bg-card overflow-hidden rounded-lg border shadow-sm shadow-foreground/5',
                className,
            )}
        >
            <button
                type="button"
                onClick={onOpen}
                className="from-brand to-brand-deep block w-full bg-gradient-to-br px-[18px] py-4 text-left text-white"
            >
                <div className="flex items-center gap-2">
                    <span className="bg-spark size-2 rounded-full ring-4 ring-[color-mix(in_srgb,var(--spark)_30%,transparent)]" />
                    <span className="text-[11px] font-bold uppercase tracking-[0.08em]">
                        Live · on the way
                    </span>
                    <span className="ml-auto text-[13px] font-semibold tabular-nums">
                        ETA {stage.etaMinutes} min
                    </span>
                </div>
                <div className="mt-3 text-xl font-bold">
                    {stage.serviceLabel} · {stage.requestRef}
                </div>
                <div className="mt-0.5 text-[13.5px] text-white/90">
                    Heading to {stage.destination}
                </div>
                <div className="mt-3.5 flex gap-1.5" aria-hidden>
                    {Array.from({ length: DASHES }).map((_, i) => (
                        <span
                            key={i}
                            className={cn(
                                'h-1 flex-1 rounded-full bg-white',
                                i < filled ? 'opacity-100' : 'opacity-35',
                            )}
                        />
                    ))}
                </div>
            </button>

            <div className="flex items-center gap-3 px-[18px] py-3.5">
                <span className="bg-secondary flex size-10 items-center justify-center rounded-full text-sm font-bold">
                    {stage.technician.initials}
                </span>
                <div className="min-w-0">
                    <div className="text-sm font-semibold">{stage.technician.name}</div>
                    <div className="text-muted-foreground flex items-center gap-1 text-xs">
                        {stage.technician.idVerified ? (
                            <BadgeCheck className="text-brand size-3.5" />
                        ) : null}
                        {stage.technician.idVerified ? 'ID verified · ' : ''}
                        {stage.technician.team}
                    </div>
                </div>
                <div className="ml-auto flex gap-2">
                    <button
                        type="button"
                        onClick={onCall}
                        aria-label="Call technician"
                        className="text-brand-strong flex size-[38px] items-center justify-center rounded-full border"
                    >
                        <Phone className="size-[17px]" />
                    </button>
                    <button
                        type="button"
                        onClick={onMessage}
                        aria-label="Message technician"
                        className="text-brand-strong flex size-[38px] items-center justify-center rounded-full border"
                    >
                        <MessageCircle className="size-[17px]" />
                    </button>
                </div>
            </div>
        </div>
    );
}
