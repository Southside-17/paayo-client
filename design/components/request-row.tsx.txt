import { Link } from '@inertiajs/react';
import { CalendarClock, ChevronRight } from 'lucide-react';

import { StatusPill, type StatusTone } from '@/components/ui/status-pill';
import { cn } from '@/lib/utils';

export type ServiceRequestSummary = {
    id: string;
    ref: string; // e.g. "SR-1048"
    title: string; // "AC cleaning"
    when: string; // "Today 2:30 PM"
    area: string; // "Makati"
    status: { tone: StatusTone; label: string };
    href?: string;
};

type Props = {
    request: ServiceRequestSummary;
    className?: string;
};

/**
 * A single service-request row for the mobile Requests list and the home
 * "Next visits" section. Links to the request detail when `href` is set.
 */
export function RequestRow({ request, className }: Props) {
    const inner = (
        <>
            <span className="bg-secondary text-muted-foreground flex size-[42px] shrink-0 items-center justify-center rounded-xl">
                <CalendarClock className="size-[19px]" />
            </span>
            <div className="min-w-0 flex-1">
                <div className="flex items-center gap-2">
                    <span className="text-[14.5px] font-semibold">{request.title}</span>
                    <span className="text-muted-foreground text-xs tabular-nums">
                        {request.ref}
                    </span>
                </div>
                <div className="text-muted-foreground mt-0.5 text-[12.5px]">
                    {request.when} · {request.area}
                </div>
                <div className="mt-2">
                    <StatusPill tone={request.status.tone} label={request.status.label} />
                </div>
            </div>
            <ChevronRight className="text-muted-foreground size-[18px]" />
        </>
    );

    const classes = cn(
        'bg-card flex w-full items-center gap-3 rounded-lg border p-3.5 text-left shadow-sm shadow-foreground/5',
        className,
    );

    return request.href ? (
        <Link href={request.href} className={classes}>
            {inner}
        </Link>
    ) : (
        <div className={classes}>{inner}</div>
    );
}
